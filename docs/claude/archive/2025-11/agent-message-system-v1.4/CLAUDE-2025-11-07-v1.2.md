// internal/adapter/inbound/job/agent_campaign_scheduler.go
type AgentCampaignScheduler struct {
    agentUseCase         ports.AgentUseCase
    logger               infrastructure.Logger
    tracingService       infrastructure.TracingService
    distributedLockMgr   infrastructure.DistributedLockManager
    
    // 排程配置
    scanInterval         time.Duration    // 掃描間隔 (30秒)
    batchSize           int              // 批次處理大小 (1000)
    maxConcurrency      int              // 最大併發數 (10)
}

// ProcessScheduledCampaigns 處理排程活動主入口
func (s *AgentCampaignScheduler) ProcessScheduledCampaigns(ctx context.Context) error {
    span, ctx := s.tracingService.StartSpan(ctx, "AgentCampaignScheduler.ProcessScheduledCampaigns")
    defer s.tracingService.SpanEnd(span)
    
    // 1. 獲取分佈式鎖確保單實例執行
    lockKey := "scheduler:agent_campaigns:process"
    mutex, err := s.distributedLockMgr.GetMutex(lockKey, infrastructure.LockOptions{
        Expiry:     5 * time.Minute,     // 5分鐘鎖過期
        Tries:      1,                   // 不重試，避免重複執行
        RetryDelay: 0,
    })
    if err != nil {
        return fmt.Errorf("failed to get scheduler lock: %w", err)
    }
    
    locked, err := mutex.TryLock()
    if err != nil || !locked {
        s.logger.Info("Scheduler already running on another instance, skipping")
        return nil
    }
    defer mutex.Unlock()
    
    // 2. 查詢到期的排程活動
    campaigns, err := s.agentUseCase.GetScheduledCampaigns(ctx, time.Now())
    if err != nil {
        s.tracingService.RecordSpanError(span, err)
        return fmt.Errorf("failed to get scheduled campaigns: %w", err)
    }
    
    if len(campaigns) == 0 {
        s.logger.Info("No scheduled campaigns found")
        return nil
    }
    
    s.logger.Info("Processing scheduled campaigns", 
        logger.Field{Key: "campaign_count", Value: len(campaigns)})
    
    // 3. 併發處理活動
    return s.processCampaignsConcurrently(ctx, campaigns)
}
```

func (u *AgentUseCase) GetScheduledCampaigns(ctx context.Context, currentTime time.Time) ([]*entity.AgentCampaign, error) {
    span, ctx := u.tracingService.StartSpan(ctx, "AgentUseCase.GetScheduledCampaigns")

    campaigns, err := u.agentCampaignRepo.GetScheduledCampaigns(ctx, currentTime)

        u.tracingService.RecordSpanError(span, err)

    }
    
    u.tracingService.TraceEvent(span, "Found scheduled campaigns", 
        attribute.Int("count", len(campaigns)))
    
    return campaigns, nil
}

// SendMessageToCampaignTargets 向活動目標代理發送訊息
func (u *AgentUseCase) SendMessageToCampaignTargets(
    ctx context.Context, 
    campaign *entity.AgentCampaign,
) (targetCount int, sentCount int, err error) {
    span, ctx := u.tracingService.StartSpan(ctx, "AgentUseCase.SendMessageToCampaignTargets",
        trace.WithAttributes(attribute.Int64("campaign.id", int64(campaign.ID))))
    defer u.tracingService.SpanEnd(span)
    
    // 1. 解析目標代理
    targetAgents, err := u.resolveTargetAgentsByType(ctx, campaign)
    if err != nil {
        u.tracingService.RecordSpanError(span, err)
        return 0, 0, fmt.Errorf("resolve target agents: %w", err)
    }
    
    // 2. 篩選活躍代理
    activeAgents, err := u.FilterActiveAgents(ctx, targetAgents)
    if err != nil {
        u.tracingService.RecordSpanError(span, err)
        return 0, 0, fmt.Errorf("filter active agents: %w", err)
    }
    
    // 3. 批量創建訊息
    sentCount, err = u.createMessagesForAgents(ctx, campaign, activeAgents)
    if err != nil {
        u.tracingService.RecordSpanError(span, err)
        return len(targetAgents), 0, fmt.Errorf("create messages: %w", err)
    }
    
    u.tracingService.TraceEvent(span, "Messages sent successfully",
        attribute.Int("target_count", len(targetAgents)),
        attribute.Int("sent_count", sentCount))
    
    return len(targetAgents), sentCount, nil
}

// FilterActiveAgents 篩選活躍代理（1個月內登入）
func (u *AgentUseCase) FilterActiveAgents(ctx context.Context, globalAgentIDs []string) ([]*entity.Agent, error) {
    span, ctx := u.tracingService.StartSpan(ctx, "AgentUseCase.FilterActiveAgents")
    defer u.tracingService.SpanEnd(span)
    
    if len(globalAgentIDs) == 0 {
        return []*entity.Agent{}, nil
    }
    
    // 批量查詢代理資訊
    agents, err := u.agentRepo.BatchGetAgentsByGlobalIDs(ctx, globalAgentIDs)
    if err != nil {
        u.tracingService.RecordSpanError(span, err)
        return nil, fmt.Errorf("batch get agents: %w", err)
    }
    
    // 篩選活躍代理（1個月內登入）
    threshold := time.Now().AddDate(0, -1, 0)
    var activeAgents []*entity.Agent
    
    for _, agent := range agents {
        if agent.CurrentSignInAt != nil && agent.CurrentSignInAt.After(threshold) {
            activeAgents = append(activeAgents, agent)
        }
    }
    
    u.tracingService.TraceEvent(span, "Active agents filtered",
        attribute.Int("total_agents", len(agents)),
        attribute.Int("active_agents", len(activeAgents)),
        attribute.String("threshold", threshold.Format(time.RFC3339)))
    
    return activeAgents, nil
}

// processCampaign 處理單個活動的完整流程
func (s *AgentCampaignScheduler) processCampaign(
    ctx context.Context, 
    campaign *entity.AgentCampaign,
) error {
    span, ctx := s.tracingService.StartSpan(ctx, "processCampaign",
        trace.WithAttributes(attribute.Int64("campaign.id", int64(campaign.ID))))
    defer s.tracingService.SpanEnd(span)
    
    s.logger.Info("Processing campaign", 
        logger.Field{Key: "campaign_id", Value: campaign.ID},
        logger.Field{Key: "title", Value: campaign.Title})
    
    //// 1. 更新狀態為 sending
    //if err := s.agentUseCase.UpdateCampaignStatus(ctx, campaign.ID, consts.AgentCampaignStatusSending); err != nil {
    //    s.tracingService.RecordSpanError(span, err)
    //    return fmt.Errorf("failed to update campaign status: %w", err)
    //}
    
    // 2. 委託UseCase執行完整發送流程
    targetCount, sentCount, err := s.agentUseCase.SendMessageToCampaignTargets(ctx, campaign)
    if err != nil {
        s.markCampaignFailed(ctx, campaign.ID, err)
        return fmt.Errorf("failed to send messages: %w", err)
    }
    
    // 3. 更新統計與完成狀態
    if err := s.agentUseCase.CompleteCampaign(ctx, campaign.ID, targetCount, sentCount); err != nil {
        return fmt.Errorf("failed to complete campaign: %w", err)
    }
    
    s.logger.Info("Campaign completed successfully",
        logger.Field{Key: "campaign_id", Value: campaign.ID},
        logger.Field{Key: "target_count", Value: targetCount},
        logger.Field{Key: "sent_count", Value: sentCount})
    
    return nil
}

// markCampaignFailed 標記活動失敗
func (s *AgentCampaignScheduler) markCampaignFailed(ctx context.Context, campaignID uint64, err error) {
    s.logger.Error("Campaign processing failed",
        logger.Field{Key: "campaign_id", Value: campaignID},
        logger.Field{Key: "error", Value: err.Error()})
    
    // 更新狀態為 failed
    if updateErr := s.agentUseCase.UpdateCampaignStatus(ctx, campaignID, consts.AgentCampaignStatusFailed); updateErr != nil {
        s.logger.Error("Failed to mark campaign as failed",
            logger.Field{Key: "campaign_id", Value: campaignID},
            logger.Field{Key: "update_error", Value: updateErr.Error()})
    }
}

// internal/adapter/inbound/job/job_registry.go
type JobRegistry struct {
    agentCampaignScheduler *job.AgentCampaignScheduler
    logger                 infrastructure.Logger
}

// RegisterJobs 註冊所有定時任務
func (jr *JobRegistry) RegisterJobs(scheduler *cron.Cron) error {
    // 每30秒執行一次代理活動排程
    _, err := scheduler.AddFunc("*/30 * * * * *", func() {
        ctx, cancel := context.WithTimeout(context.Background(), 5*time.Minute)
        defer cancel()
        
        if err := jr.agentCampaignScheduler.ProcessScheduledCampaigns(ctx); err != nil {
            jr.logger.Error("Agent campaign scheduler failed",
                logger.Field{Key: "error", Value: err.Error()})
        }
    })
    
    if err != nil {
        return fmt.Errorf("register agent campaign scheduler: %w", err)
    }
    
    jr.logger.Info("Agent campaign scheduler registered successfully")
    return nil
}

// cmd/scheduler/main.go - 排程器服務
func main() {
    // 1. 載入配置
    config, err := config.Load()
    if err != nil {
        log.Fatal("Failed to load config:", err)
    }
    
    // 2. 依賴注入
    container, err := di.InitializeSchedulerContainer(config)
    if err != nil {
        log.Fatal("Failed to initialize container:", err)
    }
    defer container.Cleanup()
    
    // 3. 創建Cron調度器
    scheduler := cron.New(cron.WithSeconds())
    
    // 4. 註冊定時任務
    if err := container.JobRegistry.RegisterJobs(scheduler); err != nil {
        log.Fatal("Failed to register jobs:", err)
    }
    
    // 5. 啟動排程器
    scheduler.Start()
    container.Logger.Info("Agent campaign scheduler started")
    
    // 6. 優雅關閉
    c := make(chan os.Signal, 1)
    signal.Notify(c, os.Interrupt, syscall.SIGTERM)
    <-c
    
    container.Logger.Info("Shutting down scheduler...")
    ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    defer cancel()
    
    scheduler.Stop()
    container.Logger.Info("Scheduler stopped gracefully")
}
```
