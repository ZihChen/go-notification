# Redis Cache 架構優化重構規劃 v1.0

**日期**: 2025-11-18  
**版本**: v1.0  
**狀態**: 規劃階段 (簡化設計，避免過度設計)  

## 概述

基於對現有Redis cache設計的分析，識別出幾個可以優化的地方。雖然目前的設計已經相當穩健且符合生產環境需求，但在錯誤處理、健康檢查、重連機制等方面仍有改進空間。

**設計原則**: 本重構計劃遵循簡潔實用的設計原則，避免過度工程化。對於Pipeline操作，保持使用Redis原生的`redis.Pipeliner`介面，不新增額外的抽象層，以確保代碼的簡潔性和可維護性。

```go
    if err != nil {
        return "", err
    }

func (m *Manager) Delete(ctx context.Context, keys ...string) error {
    client, err := m.GetClient()
    if err != nil {
        return err
    }
    return client.Del(ctx, keys...).Err()
}

func (m *Manager) Exists(ctx context.Context, keys ...string) (int64, error) {
    client, err := m.GetClient()
    if err != nil {
        return 0, err
    }
    return client.Exists(ctx, keys...).Result()
}

func (m *Manager) MSet(ctx context.Context, pairs map[string]interface{}, expiration time.Duration) error {
    pipeline, err := m.Pipeline()
    if err != nil {
        return err
    }
    
    for key, value := range pairs {
        pipeline.Set(ctx, key, value, expiration)
    }
    
    _, err = pipeline.Exec(ctx)
    return err
}

func (m *Manager) SetEX(ctx context.Context, key string, value interface{}, expiration time.Duration) error {
    _, err := m.Set(ctx, key, value, expiration)
    return err
}
```

```go
// internal/infrastructure/cache/redis/manager.go
func (m *Manager) HealthCheck() bool {
    m.mutex.RLock()
    defer m.mutex.RUnlock()
    
    if m.client == nil {
        return false
    }
    
    // 使用短超時進行ping檢查
    ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
    defer cancel()
    
    // 執行ping命令檢查連通性
    if err := m.client.Ping(ctx).Err(); err != nil {
        // 記錄健康檢查失敗
        if m.logger != nil {
            m.logger.Warn("Redis health check failed", 
                zap.Error(err),
                zap.String("address", m.config.Addr))
        }
        return false
    }
    
    return true
}

// HealthCheckWithDetails 提供詳細的健康狀態資訊
func (m *Manager) HealthCheckWithDetails() (bool, map[string]interface{}) {
    details := make(map[string]interface{})
    
    m.mutex.RLock()
    defer m.mutex.RUnlock()
    
    if m.client == nil {
        details["status"] = "client_not_initialized"
        return false, details
    }
    
    ctx, cancel := context.WithTimeout(context.Background(), 2*time.Second)
    defer cancel()
    
    start := time.Now()
    err := m.client.Ping(ctx).Err()
    latency := time.Since(start)
    
    details["ping_latency_ms"] = latency.Milliseconds()
    details["redis_addr"] = m.config.Addr
    
    if err != nil {
        details["status"] = "ping_failed"
        details["error"] = err.Error()
        return false, details
    }
    
    // 檢查連接池狀態
    if poolStats := m.client.PoolStats(); poolStats != nil {
        details["pool_hits"] = poolStats.Hits
        details["pool_misses"] = poolStats.Misses
        details["pool_timeouts"] = poolStats.Timeouts
        details["pool_total_conns"] = poolStats.TotalConns
        details["pool_idle_conns"] = poolStats.IdleConns
    }
    
    details["status"] = "healthy"
    return true, details
}
```

```go
    if err != nil {
        return "", fmt.Errorf("failed to get redis client: %w", err)
    }

// GetStats 獲取緩存統計資訊
func (m *Manager) GetStats() CacheStats {
    m.mutex.RLock()
    defer m.mutex.RUnlock()
    
    stats := CacheStats{
        Metadata: make(map[string]string),
    }
    
    if m.client == nil {
        stats.LastError = "client not initialized"
        return stats
    }
    
    poolStats := m.client.PoolStats()
    if poolStats != nil {
        stats.Hits = poolStats.Hits
        stats.Misses = poolStats.Misses
        stats.Timeouts = poolStats.Timeouts
        stats.TotalConns = poolStats.TotalConns
        stats.IdleConns = poolStats.IdleConns
    }
    
    // 測量延遲
    start := time.Now()
    ctx, cancel := context.WithTimeout(context.Background(), time.Second)
    defer cancel()
    
    if err := m.client.Ping(ctx).Err(); err != nil {
        stats.LastError = err.Error()
    }
    stats.Latency = time.Since(start)
    
    stats.Metadata["redis_version"] = m.client.Info(ctx, "server").Val()
    
    return stats
}
```

```go
// test/integration/concurrent_test.go
func TestManager_ConcurrentOperations(t *testing.T) {
    manager := setupTestManager(t)
    
    t.Run("高併發讀寫測試", func(t *testing.T) {
        const (
            goroutines = 100
            operations = 1000
        )
        
        var wg sync.WaitGroup
        errors := make(chan error, goroutines*operations)
        
        // 並發寫入
        for i := 0; i < goroutines; i++ {
            wg.Add(1)
            go func(id int) {
                defer wg.Done()
                
                for j := 0; j < operations; j++ {
                    key := fmt.Sprintf("key_%d_%d", id, j)
                    value := fmt.Sprintf("value_%d_%d", id, j)
                    
                    if err := manager.Set(context.Background(), key, value, time.Minute); err != nil {
                        errors <- err
                    }
                }
            }(i)
        }
        
        // 並發讀取
        for i := 0; i < goroutines; i++ {
            wg.Add(1)
            go func(id int) {
                defer wg.Done()
                
                for j := 0; j < operations; j++ {
                    key := fmt.Sprintf("key_%d_%d", id, j)
                    
                    if _, err := manager.Get(context.Background(), key); err != nil {
                        // 可能的not found錯誤是正常的
                        if !strings.Contains(err.Error(), "key not found") {
                            errors <- err
                        }
                    }
                }
            }(i)
        }
        
        wg.Wait()
        close(errors)
        
        // 檢查錯誤
        var errorList []error
        for err := range errors {
            errorList = append(errorList, err)
        }
        
        if len(errorList) > 0 {
            t.Logf("發現 %d 個錯誤", len(errorList))
            for _, err := range errorList[:min(10, len(errorList))] {
                t.Logf("錯誤: %v", err)
            }
        }
        
        // 容錯率測試（允許少量錯誤）
        errorRate := float64(len(errorList)) / float64(goroutines*operations*2)
        assert.Less(t, errorRate, 0.01, "錯誤率應小於1%")
    })
}
```
