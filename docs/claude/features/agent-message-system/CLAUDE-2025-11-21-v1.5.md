```go
func parseStatusFilter(statusStr string) []string {
    if statusStr == "" {
        return nil
    }
    statuses := strings.Split(statusStr, ",")
    result := make([]string, 0, len(statuses))
    for _, status := range statuses {
        if trimmed := strings.TrimSpace(status); trimmed != "" {
            result = append(result, trimmed)
        }
    }
    return result
}
```

**Repository層事務處理** (`internal/adapter/outbound/repository/agent/agent_campaign_repository.go`):
```go
func (r *AgentCampaignRepository) BatchDelete(ctx context.Context, ids []uint64) error {
    if len(ids) == 0 {
        return fmt.Errorf("no IDs provided for batch delete")
    }

    // 使用事務確保批量操作的原子性
    tx := r.db.WithContext(ctx).Begin()
    defer func() {
        if r := recover(); r != nil {
            tx.Rollback()
        }
    }()

    // 先批量更新狀態為 cancelled
    updateResult := tx.Model(&models.AgentCampaign{}).
        Where("id IN ?", ids).
        Update("status", "cancelled")
    if updateResult.Error != nil {
        tx.Rollback()
        return fmt.Errorf("batch update campaigns status to cancelled failed: %w", updateResult.Error)
    }

    // 批量軟刪除
    deleteResult := tx.Where("id IN ?", ids).Delete(&models.AgentCampaign{})
    if deleteResult.Error != nil {
        tx.Rollback()
        return fmt.Errorf("batch delete agent campaigns failed: %w", deleteResult.Error)
    }

    return tx.Commit().Error
}
```

**UseCase層業務邏輯** (`internal/application/usecase/agent/agent_usecase.go`):
```go
func (u *AgentUseCase) BatchDeleteAgentCampaigns(
    ctx context.Context,
    req *dto.BatchDeleteAgentCampaignsRequest,
) error {
    if len(req.IDs) == 0 {
        return fmt.Errorf("no IDs provided for batch delete")
    }

    // 驗證ID有效性
    for _, id := range req.IDs {
        if id == 0 {
            return fmt.Errorf("invalid campaign ID: %d", id)
        }
    }

    // 執行批量刪除
    return u.agentCampaignRepo.BatchDelete(ctx, req.IDs)
}
```

**API Handler端點** (`internal/adapter/inbound/handler/api/agent_handler.go`):
```go
// BatchDeleteAgentCampaigns 批量刪除代理訊息活動
// @Summary 批量刪除代理訊息活動
// @Description 根據ID列表批量刪除代理訊息活動，支援高效能批量操作
// @Tags 代理訊息活動
// @Accept json
// @Produce json
// @Param request body dto.BatchDeleteAgentCampaignsRequest true "批量刪除請求"
// @Success 200 {object} SuccessResponse
// @Failure 400 {object} ErrorResponse
// @Failure 500 {object} ErrorResponse
// @Security ApiKeyAuth
// @Router /api/v1/agent-campaigns [delete]
func (h *AgentHandler) BatchDeleteAgentCampaigns(c *gin.Context) {
    var req dto.BatchDeleteAgentCampaignsRequest
    if err := c.ShouldBindJSON(&req); err != nil {
        response.BadRequest(c, "invalid request payload", err.Error()).Return()
        return
    }

    // 執行批量刪除
    if err := h.agentUseCase.BatchDeleteAgentCampaigns(c.Request.Context(), &req); err != nil {
        h.logger.ErrorWithContext(
            c.Request.Context(),
            "failed to batch delete agent campaigns",
            h.logger.Error("err", err),
        )
        response.InternalServerError(c, "failed to batch delete agent campaigns", err.Error()).Return()
        return
    }

    response.DeletedSuccess(c).Return()
}
```

## 效能特性

### 批量刪除效能優勢
1. **事務原子性**: 全部成功或全部失敗，確保資料一致性
2. **批量SQL操作**: 使用`WHERE id IN (?)`減少資料庫往返次數
3. **兩階段處理**: 先更新狀態再軟刪除，符合業務邏輯
4. **錯誤恢復**: 自動回滾機制確保操作安全性

### 狀態篩選效能優化
1. **字串解析優化**: 一次性解析逗號分隔字串
2. **記憶體效率**: 避免不必要的字串分配
3. **SQL優化**: 直接使用IN查詢，避免多次單條查詢
