## 📋 實作總覽

### 功能範圍

透過 Server-Sent Events (SSE) 協議實現實時推播通知系統,支援:

- ✅ **後端推播 API** (REST + API Key 認證)
  - 廣播推送 (所有線上玩家)
  - 個別推送 (特定玩家清單)
  - 統計資訊查詢

- ✅ **前端接收 API** (SSE + JWT 認證)
  - SSE 長連接接收實時訊息
  - 查詢歷史訊息
  - 標記已讀
  - 刪除訊息

```go
package entity

import (
    "time"

// SSENotification 實時推播通知實體
type SSENotification struct {
    ID           string            `json:"id" gorm:"primaryKey;type:varchar(36)"` // UUID v4
    PlayerID     *string           `json:"player_id,omitempty" gorm:"type:varchar(64);index"` // NULL = 廣播
    Title        string            `json:"title" gorm:"type:varchar(100);not null"`
    Message      string            `json:"message" gorm:"type:varchar(500);not null"`
    Type         string            `json:"type" gorm:"type:enum('activity','promotion','system','announcement');not null"`
    Priority     string            `json:"priority" gorm:"type:enum('low','medium','high','critical');not null"`
    IconURL      *string           `json:"icon_url,omitempty" gorm:"type:varchar(500)"`
    ActionURL    *string           `json:"action_url,omitempty" gorm:"type:varchar(500)"`
    Visibility   string            `json:"visibility" gorm:"type:enum('visible','hidden');not null;default:'visible'"`
    IsRead       bool              `json:"is_read" gorm:"default:false"`
    CreatedAt    time.Time         `json:"created_at" gorm:"not null;index"`
    ExpiresAt    time.Time         `json:"expires_at" gorm:"not null;index"`
    Metadata     map[string]string `json:"metadata,omitempty" gorm:"type:json"`
}

// TableName 指定資料表名稱
func (SSENotification) TableName() string {
    return "sse_notifications"
}

// IsExpired 檢查是否過期
func (n *SSENotification) IsExpired() bool {
    return time.Now().After(n.ExpiresAt)
}

// IsBroadcast 檢查是否為廣播訊息
func (n *SSENotification) IsBroadcast() bool {
    return n.PlayerID == nil
}

// NotificationStats 通知統計實體
type NotificationStats struct {
    ID              uint64    `json:"id" gorm:"primaryKey;autoIncrement"`
    Date            string    `json:"date" gorm:"type:date;not null;index:idx_date_type_priority,priority:1"`
    Type            string    `json:"type" gorm:"type:varchar(50);not null;index:idx_date_type_priority,priority:2"`
    Priority        string    `json:"priority" gorm:"type:varchar(20);not null;index:idx_date_type_priority,priority:3"`
    TotalSent       int64     `json:"total_sent" gorm:"not null;default:0"`
    TotalDelivered  int64     `json:"total_delivered" gorm:"not null;default:0"`
    CreatedAt       time.Time `json:"created_at" gorm:"not null"`
    UpdatedAt       time.Time `json:"updated_at" gorm:"not null"`
}

// TableName 指定資料表名稱
func (NotificationStats) TableName() string {
    return "sse_notification_stats"
}

// DeliveryRate 計算遞送率
func (s *NotificationStats) DeliveryRate() float64 {
    if s.TotalSent == 0 {
        return 0
    }
    return float64(s.TotalDelivered) / float64(s.TotalSent) * 100
}
```

```go
package sse_notification

import (
    "context"
    "fmt"
    "time"

    "github.com/google/uuid"
    "github.com/jvdiamondtech/ms-notification-cat/internal/application/dto"
    "github.com/jvdiamondtech/ms-notification-cat/internal/domain/consts"
    "github.com/jvdiamondtech/ms-notification-cat/internal/domain/entity"
    "github.com/jvdiamondtech/ms-notification-cat/internal/domain/errmsg"
    "github.com/jvdiamondtech/ms-notification-cat/internal/domain/ports/inbound"
    "github.com/jvdiamondtech/ms-notification-cat/internal/domain/ports/outbound/infrastructure"
    "github.com/jvdiamondtech/ms-notification-cat/internal/domain/ports/outbound/repository"
    "github.com/jvdiamondtech/ms-notification-cat/internal/domain/ports/outbound/service"
)

type sseNotificationUseCase struct {
    repo         repository.SSENotificationRepository
    sseManager   service.SSEManager
    eventService service.EventService // KDS 審計日誌
    logger       infrastructure.Logger
}

// NewSSENotificationUseCase 創建 SSE 通知使用案例
func NewSSENotificationUseCase(
    repo repository.SSENotificationRepository,
    sseManager service.SSEManager,
    eventService service.EventService,
    logger infrastructure.Logger,
) inbound.SSENotificationUseCase {
    return &sseNotificationUseCase{
        repo:         repo,
        sseManager:   sseManager,
        eventService: eventService,
        logger:       logger,
    }
}

// BroadcastNotification 廣播推送實現
func (uc *sseNotificationUseCase) BroadcastNotification(
    ctx context.Context,
    req *dto.BroadcastRequest,
) (*dto.BroadcastResponse, error) {
    // 1. 創建通知實體 (playerID = nil 表示廣播)
    notification := &entity.SSENotification{
        ID:         uuid.New().String(),
        PlayerID:   nil, // 廣播
        Title:      req.Title,
        Message:    req.Message,
        Type:       req.Type,
        Priority:   req.Priority,
        IconURL:    req.IconURL,
        ActionURL:  req.ActionURL,
        Visibility: req.Visibility,
        CreatedAt:  time.Now(),
        ExpiresAt:  req.ExpiryAt,
        Metadata:   req.Metadata,
    }

    // 2. 持久化訊息到 MySQL
    if err := uc.repo.Create(ctx, notification); err != nil {
        uc.logger.ErrorWithContext(ctx, "failed to create broadcast notification",
            uc.logger.Error("err", err),
            uc.logger.String("notification_id", notification.ID))
        return nil, fmt.Errorf("create notification: %w", err)
    }

    // 3. 廣播到所有線上玩家 (透過 SSE Manager)
    queuedCount, err := uc.sseManager.BroadcastToAll(ctx, notification)
    if err != nil {
        uc.logger.ErrorWithContext(ctx, "failed to broadcast notification",
            uc.logger.Error("err", err),
            uc.logger.String("notification_id", notification.ID))
        // 廣播失敗不返回錯誤,訊息已持久化
    }

    // 4. 更新統計資訊
    if err := uc.repo.IncrementStats(ctx,
        time.Now().Format("2006-01-02"),
        notification.Type,
        notification.Priority,
        int64(queuedCount), // sent
        int64(queuedCount), // delivered
    ); err != nil {
        uc.logger.ErrorWithContext(ctx, "failed to update stats", uc.logger.Error("err", err))
    }

    // 5. 記錄審計日誌到 KDS
    uc.eventService.PublishEvent(ctx, "notification.broadcast", map[string]interface{}{
        "notification_id": notification.ID,
        "merchant_id":     req.GlobalMerchantID,
        "queued_count":    queuedCount,
        "type":            notification.Type,
        "priority":        notification.Priority,
    })

    // 6. 返回回應
    return &dto.BroadcastResponse{
        BroadcastID:  notification.ID,
        TotalPlayers: queuedCount,
        QueuedCount:  queuedCount,
        SentAt:       notification.CreatedAt,
        Status:       consts.NotificationStatusQueued,
    }, nil
}

// SendNotification 個別推送實現
func (uc *sseNotificationUseCase) SendNotification(
    ctx context.Context,
    req *dto.SendNotificationRequest,
) (*dto.SendNotificationResponse, error) {
    // 1. 驗證玩家 ID 數量
    if len(req.PlayerIDs) > consts.MaxPlayerIDsPerRequest {
        return nil, fmt.Errorf("player_ids exceeds maximum limit of %d", consts.MaxPlayerIDsPerRequest)
    }

    sendID := uuid.New().String()
    queuedCount := 0

    // 2. 為每位玩家創建通知
    notifications := make([]*entity.SSENotification, 0, len(req.PlayerIDs))
    for _, playerID := range req.PlayerIDs {
        playerIDCopy := playerID // 避免閉包問題
        notification := &entity.SSENotification{
            ID:         uuid.New().String(),
            PlayerID:   &playerIDCopy,
            Title:      req.Title,
            Message:    req.Message,
            Type:       req.Type,
            Priority:   req.Priority,
            IconURL:    req.IconURL,
            ActionURL:  req.ActionURL,
            Visibility: req.Visibility,
            CreatedAt:  time.Now(),
            ExpiresAt:  req.ExpiryAt,
            Metadata:   req.Metadata,
        }
        notifications = append(notifications, notification)
    }

    // 3. 批次持久化到 MySQL
    if err := uc.repo.BatchCreate(ctx, notifications); err != nil {
        uc.logger.ErrorWithContext(ctx, "failed to batch create notifications",
            uc.logger.Error("err", err),
            uc.logger.String("send_id", sendID))
        return nil, fmt.Errorf("batch create notifications: %w", err)
    }

    // 4. 推送給線上玩家,離線玩家加入離線佇列
    for _, notification := range notifications {
        isOnline, _ := uc.sseManager.IsPlayerOnline(ctx, *notification.PlayerID)
        if isOnline {
            if err := uc.sseManager.SendToPlayer(ctx, *notification.PlayerID, notification); err != nil {
                uc.logger.WarnWithContext(ctx, "failed to send to online player",
                    uc.logger.Error("err", err),
                    uc.logger.String("player_id", *notification.PlayerID))
                // 線上推送失敗,加入離線佇列
                uc.sseManager.EnqueueOfflineMessage(ctx, *notification.PlayerID, notification)
            } else {
                queuedCount++
            }
        } else {
            // 離線玩家,加入離線佇列
            if err := uc.sseManager.EnqueueOfflineMessage(ctx, *notification.PlayerID, notification); err != nil {
                uc.logger.ErrorWithContext(ctx, "failed to enqueue offline message",
                    uc.logger.Error("err", err),
                    uc.logger.String("player_id", *notification.PlayerID))
            }
        }
    }

    // 5. 更新統計資訊
    uc.repo.IncrementStats(ctx,
        time.Now().Format("2006-01-02"),
        req.Type,
        req.Priority,
        int64(len(req.PlayerIDs)), // sent
        int64(queuedCount),         // delivered
    )

    // 6. 記錄審計日誌到 KDS
    uc.eventService.PublishEvent(ctx, "notification.send", map[string]interface{}{
        "send_id":       sendID,
        "merchant_id":   req.GlobalMerchantID,
        "player_count":  len(req.PlayerIDs),
        "queued_count":  queuedCount,
        "type":          req.Type,
        "priority":      req.Priority,
    })

    // 7. 返回回應
    return &dto.SendNotificationResponse{
        SendID:      sendID,
        PlayerCount: len(req.PlayerIDs),
        QueuedCount: queuedCount,
        SentAt:      time.Now(),
        Status:      consts.NotificationStatusQueued,
    }, nil
}

// GetNotificationStats 查詢統計資訊實現
func (uc *sseNotificationUseCase) GetNotificationStats(
    ctx context.Context,
    date string,
) (*dto.NotificationStatsResponse, error) {
    // 1. 查詢統計資料
    stats, err := uc.repo.GetStatsByDate(ctx, date)
    if err != nil {
        uc.logger.ErrorWithContext(ctx, "failed to get stats", uc.logger.Error("err", err))
        return nil, fmt.Errorf("get stats: %w", err)
    }

    // 2. 聚合統計資料
    response := &dto.NotificationStatsResponse{
        Date:       date,
        ByType:     make(map[string]int),
        ByPriority: make(map[string]int),
    }

    var totalSent, totalDelivered int64
    for _, stat := range stats {
        totalSent += stat.TotalSent
        totalDelivered += stat.TotalDelivered
        response.ByType[stat.Type] += int(stat.TotalSent)
        response.ByPriority[stat.Priority] += int(stat.TotalSent)
    }

    response.TotalSent = totalSent
    response.TotalDelivered = totalDelivered
    if totalSent > 0 {
        response.DeliveryRate = float64(totalDelivered) / float64(totalSent) * 100
    }

    // 3. 查詢線上玩家數
    onlineCount, _ := uc.sseManager.GetOnlinePlayerCount(ctx)
    response.OnlinePlayers = onlineCount

    // 4. 查詢總玩家數
    totalPlayers, _ := uc.repo.GetTotalPlayerCount(ctx)
    response.TotalPlayers = totalPlayers

    return response, nil
}

// StreamNotifications SSE 長連接實現
func (uc *sseNotificationUseCase) StreamNotifications(
    ctx context.Context,
    playerID string,
    writer inbound.SSEWriter,
) error {
    // 1. 註冊連接
    if err := uc.sseManager.RegisterConnection(ctx, playerID, writer); err != nil {
        return fmt.Errorf("register connection: %w", err)
    }
    defer uc.sseManager.UnregisterConnection(ctx, playerID)

    // 2. 發送歡迎訊息
    writer.Write("connected", map[string]interface{}{
        "player_id": playerID,
        "timestamp": time.Now(),
    })

    // 3. 發送離線訊息
    offlineMessages, err := uc.sseManager.GetOfflineMessages(ctx, playerID)
    if err != nil {
        uc.logger.ErrorWithContext(ctx, "failed to get offline messages",
            uc.logger.Error("err", err),
            uc.logger.String("player_id", playerID))
    } else {
        for _, msg := range offlineMessages {
            writer.Write("notification", msg)
        }
        // 清除已發送的離線訊息
        uc.sseManager.ClearOfflineMessages(ctx, playerID)
    }

    // 4. 保持連接 (等待 context 取消)
    <-ctx.Done()

    return nil
}

// GetNotificationHistory 查詢歷史訊息實現
func (uc *sseNotificationUseCase) GetNotificationHistory(
    ctx context.Context,
    playerID string,
```

```go
    "github.com/gin-gonic/gin"
    "github.com/jvdiamondtech/ms-notification-cat/internal/domain/ports/inbound"
)

// GinSSEWriter Gin SSE Writer 實作
type GinSSEWriter struct {
    writer  gin.ResponseWriter
    flusher http.Flusher
    ctx     *gin.Context
}

// NewGinSSEWriter 創建 Gin SSE Writer
func NewGinSSEWriter(writer gin.ResponseWriter, ctx *gin.Context) inbound.SSEWriter {
    flusher, ok := writer.(http.Flusher)
    if !ok {
        // 如果不支援 Flusher,使用 nil
        flusher = nil
    }

    return &GinSSEWriter{
        writer:  writer,
        flusher: flusher,
        ctx:     ctx,
    }
}

// Write 寫入 SSE 事件
func (w *GinSSEWriter) Write(event string, data interface{}) error {
    // 將資料序列化為 JSON
    jsonData, err := json.Marshal(data)
    if err != nil {
        return fmt.Errorf("marshal data: %w", err)
    }

    // 寫入 SSE 格式
    // event: <event_name>
    // data: <json_data>
    // \n
    if _, err := fmt.Fprintf(w.writer, "event: %s\n", event); err != nil {
        return fmt.Errorf("write event: %w", err)
    }

    if _, err := fmt.Fprintf(w.writer, "data: %s\n\n", string(jsonData)); err != nil {
        return fmt.Errorf("write data: %w", err)
    }

    // 立即刷新緩衝區
    return w.Flush()
}

// Flush 刷新緩衝區
func (w *GinSSEWriter) Flush() error {
    if w.flusher != nil {
        w.flusher.Flush()
    }
    return nil
}

// Close 關閉連接
func (w *GinSSEWriter) Close() error {
    // Gin 的連接由框架管理,這裡不需要特別關閉
    return nil
}
```

```go
// RegisterConnection 註冊玩家 SSE 連接
func (m *sseManager) RegisterConnection(ctx context.Context, playerID string, writer inbound.SSEWriter) error {
    m.mu.Lock()
    defer m.mu.Unlock()

    // 加入本地連接池
    m.connections[playerID] = writer

    // 註冊到 Redis (用於跨實例狀態同步)
    key := consts.RedisKeySSEOnlinePrefix + playerID
    if err := m.redisClient.Set(ctx, key, "1", 0).Err(); err != nil {
        m.logger.ErrorWithContext(ctx, "failed to register to redis",
            m.logger.Error("err", err),
            m.logger.String("player_id", playerID))
        return fmt.Errorf("redis set: %w", err)
    }

    m.logger.InfoWithContext(ctx, "player connected",
        m.logger.String("player_id", playerID))

    return nil
}

// UnregisterConnection 取消註冊玩家 SSE 連接
func (m *sseManager) UnregisterConnection(ctx context.Context, playerID string) error {
    m.mu.Lock()
    defer m.mu.Unlock()

    // 從本地連接池移除
    delete(m.connections, playerID)

    // 從 Redis 移除
    key := consts.RedisKeySSEOnlinePrefix + playerID
    if err := m.redisClient.Del(ctx, key).Err(); err != nil {
        m.logger.ErrorWithContext(ctx, "failed to unregister from redis",
            m.logger.Error("err", err),
            m.logger.String("player_id", playerID))
        return fmt.Errorf("redis del: %w", err)
    }

    m.logger.InfoWithContext(ctx, "player disconnected",
        m.logger.String("player_id", playerID))

    return nil
}

// IsPlayerOnline 檢查玩家是否在線
func (m *sseManager) IsPlayerOnline(ctx context.Context, playerID string) (bool, error) {
    // 先檢查本地連接池
    m.mu.RLock()
    _, existsLocal := m.connections[playerID]
    m.mu.RUnlock()

    if existsLocal {
        return true, nil
    }

    // 檢查 Redis (可能在其他實例上)
    key := consts.RedisKeySSEOnlinePrefix + playerID
    exists, err := m.redisClient.Exists(ctx, key).Result()
    if err != nil {
        return false, fmt.Errorf("redis exists: %w", err)
    }

    return exists > 0, nil
}

// GetOnlinePlayerCount 獲取線上玩家數量
func (m *sseManager) GetOnlinePlayerCount(ctx context.Context) (int, error) {
    // 使用 Redis SCAN 查詢所有線上玩家
    pattern := consts.RedisKeySSEOnlinePrefix + "*"
    var cursor uint64
    var count int

    for {
        keys, newCursor, err := m.redisClient.Scan(ctx, cursor, pattern, 100).Result()
        if err != nil {
            return 0, fmt.Errorf("redis scan: %w", err)
        }

        count += len(keys)
        cursor = newCursor

        if cursor == 0 {
            break
        }
    }

    return count, nil
}

// GetOnlinePlayerIDs 獲取所有線上玩家 ID 列表
func (m *sseManager) GetOnlinePlayerIDs(ctx context.Context) ([]string, error) {
    pattern := consts.RedisKeySSEOnlinePrefix + "*"
    var cursor uint64
    var playerIDs []string

    for {
        keys, newCursor, err := m.redisClient.Scan(ctx, cursor, pattern, 100).Result()
        if err != nil {
            return nil, fmt.Errorf("redis scan: %w", err)
        }

        // 提取 playerID (移除 prefix)
        prefix := consts.RedisKeySSEOnlinePrefix
        for _, key := range keys {
            playerID := key[len(prefix):]
            playerIDs = append(playerIDs, playerID)
        }

        cursor = newCursor
        if cursor == 0 {
            break
        }
    }

    return playerIDs, nil
}

// ========== 訊息推送 ==========

// BroadcastToAll 廣播推送給所有線上玩家
func (m *sseManager) BroadcastToAll(ctx context.Context, notification *entity.SSENotification) (int, error) {
    m.mu.RLock()
    defer m.mu.RUnlock()

    successCount := 0

    // 遍歷本地連接池廣播
    for playerID, writer := range m.connections {
        if err := writer.Write("notification", notification); err != nil {
            m.logger.WarnWithContext(ctx, "failed to send to player",
                m.logger.Error("err", err),
                m.logger.String("player_id", playerID),
                m.logger.String("notification_id", notification.ID))
            continue
        }
        successCount++
    }

    m.logger.InfoWithContext(ctx, "broadcast completed",
        m.logger.Int("total_players", len(m.connections)),
        m.logger.Int("success_count", successCount),
        m.logger.String("notification_id", notification.ID))

    return successCount, nil
}

// SendToPlayer 推送給單一玩家
func (m *sseManager) SendToPlayer(ctx context.Context, playerID string, notification *entity.SSENotification) error {
    m.mu.RLock()
    writer, exists := m.connections[playerID]
    m.mu.RUnlock()

    if !exists {
        return fmt.Errorf("player not connected: %s", playerID)
    }

    if err := writer.Write("notification", notification); err != nil {
        return fmt.Errorf("write notification: %w", err)
    }

    return nil
}

// SendToPlayers 批次推送給多位玩家
func (m *sseManager) SendToPlayers(ctx context.Context, playerIDs []string, notification *entity.SSENotification) (int, error) {
    m.mu.RLock()
    defer m.mu.RUnlock()

    successCount := 0

    for _, playerID := range playerIDs {
        writer, exists := m.connections[playerID]
        if !exists {
            continue
        }

        if err := writer.Write("notification", notification); err != nil {
            m.logger.WarnWithContext(ctx, "failed to send to player",
                m.logger.Error("err", err),
                m.logger.String("player_id", playerID))
            continue
        }

        successCount++
    }

    return successCount, nil
}

// ========== 離線訊息管理 ==========

// EnqueueOfflineMessage 將訊息加入離線佇列 (Redis Streams)
func (m *sseManager) EnqueueOfflineMessage(ctx context.Context, playerID string, notification *entity.SSENotification) error {
    key := consts.RedisKeySSEOfflinePrefix + playerID

    // 序列化通知為 JSON
    data, err := json.Marshal(notification)
    if err != nil {
        return fmt.Errorf("marshal notification: %w", err)
    }

    // 使用 Redis Streams 儲存
    _, err = m.redisClient.XAdd(ctx, &redis.XAddArgs{
        Stream: key,
        MaxLen: 100, // 最多保留 100 條離線訊息
        Values: map[string]interface{}{
            "notification": string(data),
            "created_at":   time.Now().Unix(),
        },
    }).Result()

    if err != nil {
        return fmt.Errorf("redis xadd: %w", err)
    }

    // 設置 TTL (7 天)
    m.redisClient.Expire(ctx, key, consts.OfflineMessageTTL)

    return nil
}

// GetOfflineMessages 獲取玩家的離線訊息
func (m *sseManager) GetOfflineMessages(ctx context.Context, playerID string) ([]*entity.SSENotification, error) {
    key := consts.RedisKeySSEOfflinePrefix + playerID

    // 讀取所有離線訊息
    messages, err := m.redisClient.XRange(ctx, key, "-", "+").Result()
    if err != nil {
        return nil, fmt.Errorf("redis xrange: %w", err)
    }

    var notifications []*entity.SSENotification

    for _, msg := range messages {
        dataStr, ok := msg.Values["notification"].(string)
        if !ok {
            continue
        }

        var notification entity.SSENotification
        if err := json.Unmarshal([]byte(dataStr), &notification); err != nil {
            m.logger.WarnWithContext(ctx, "failed to unmarshal offline message",
                m.logger.Error("err", err))
            continue
        }

        notifications = append(notifications, &notification)
    }

    return notifications, nil
}

// ClearOfflineMessages 清除玩家的離線訊息
func (m *sseManager) ClearOfflineMessages(ctx context.Context, playerID string) error {
    key := consts.RedisKeySSEOfflinePrefix + playerID
    return m.redisClient.Del(ctx, key).Err()
}
```

- **後端 API** (`/admin/*`): 使用 API Key 認證
- **前端 API** (`/notifications/*`): 使用 JWT 認證
- 不同的 middleware 處理不同的認證機制

### 3. 錯誤處理規範
- 所有錯誤使用 `internal/domain/errmsg/errors.go` 定義
- 使用 `fmt.Errorf` 包裝錯誤,保留錯誤鏈
- 日誌記錄使用 `infrastructure.Logger`
- HTTP 錯誤統一使用 `response` 工具包

### 4. 測試策略
- **單元測試**: UseCase 層 100% 覆蓋
- **Mock 測試**: Repository 層使用 testify/mock
- **整合測試**: Handler 層整合測試
- **壓力測試**: wrk/ab 效能測試

### 5. 效能目標
| 指標 | 目標 | 驗收標準 |
|------|------|----------|
| 廣播吞吐量 | > 10,000 QPS | wrk 測試 |
| 訊息入隊延遲 | < 100ms (P99) | 壓力測試 |
| 並發連接數 | 10,000+ | SSE 連接測試 |
| 訊息遞送率 | 99% | 統計資料驗證 |

### 6. 安全性要求
- ✅ API Key 驗證正確實施
- ✅ JWT Token 驗證與過期處理
- ✅ 所有推播事件記錄至 KDS (審計日誌)
- ✅ 敏感操作記錄完整
- ✅ 實施速率限制 (防止濫用)

### 7. 監控與告警
- **關鍵指標**:
  - 線上玩家數
  - 訊息推送成功率
  - SSE 連接異常斷開率
  - Redis/MySQL 連接池使用率
  - API 回應時間 (P50/P95/P99)
