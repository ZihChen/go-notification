**Version**: v1.0
**Date**: 2026-01-27
**Status**: 規劃中
**Reference**: `fat-event-cat/internal/infrastructure/metrics/metrics.go`

## 1. 概述

將 OpenTelemetry Metrics 服務導入 fat-notification-cat，提供系統運行時的指標收集與監控能力。

### 1.1 目標

- 收集 API 請求指標（請求數、延遲、錯誤率）
- 收集訊息處理指標（Campaign 處理、訊息發送、Worker 處理）
- 收集 KDS Consumer 指標（事件接收、處理延遲）
- 支援 OTLP HTTP 協議導出至 OpenObserve 或其他後端

```go
type MetricsConfig struct {

}

// OTLPMetricsConfig OTLP Metrics 配置
type OTLPMetricsConfig struct {
    Enabled  bool              `mapstructure:"enabled"`
    Endpoint string            `mapstructure:"endpoint"`  // e.g., "https://openobserve.example.com"
    Path     string            `mapstructure:"path"`      // e.g., "/api/default/v1/metrics"
    Insecure bool              `mapstructure:"insecure"`  // 是否使用 HTTP（非 HTTPS）
    Headers  map[string]string `mapstructure:"headers"`   // 自定義 Headers（如認證）
    Interval time.Duration     `mapstructure:"interval"`  // 導出間隔，預設 60s
    Timeout  time.Duration     `mapstructure:"timeout"`   // 請求超時，預設 10s
}
```

```go
// Metrics OpenTelemetry Metrics 服務
type Metrics struct {
    // API Metrics
    apiRequests  metric.Int64Counter     // API 請求計數

    apiErrors    metric.Int64Counter     // API 錯誤計數

    campaignProcessed metric.Int64Counter     // Campaign 處理計數
    campaignDuration  metric.Float64Histogram // Campaign 處理延遲

    // Message Metrics
    messageSent       metric.Int64Counter // 訊息發送計數
    messageDelivered  metric.Int64Counter // 訊息送達計數
    messageFailed     metric.Int64Counter // 訊息失敗計數

    // Consumer Metrics
    eventReceived     metric.Int64Counter     // KDS 事件接收計數
    eventProcessed    metric.Int64Counter     // 事件處理計數
    eventDuration     metric.Float64Histogram // 事件處理延遲

    taskProcessed     metric.Int64Counter     // 任務處理計數
    taskDuration      metric.Float64Histogram // 任務處理延遲

    logger        *logger.Logger
```

### 2.4 核心方法設計

#### 2.4.1 初始化方法

```go
// NewMetrics 建立新的 Metrics 服務
func NewMetrics(cfg *config.Config, log *logger.Logger) (*Metrics, error) {
    m := &Metrics{
        logger:  log,
        enabled: cfg.Metrics.Enabled && cfg.Metrics.OTLP.Enabled,
    }

    if !m.enabled {
        log.Info("metrics is disabled")
        return m, nil
    }

    // 建立 OTLP HTTP Exporter
    exporter, err := m.createExporter(cfg)
    if err != nil {
        return nil, fmt.Errorf("failed to create OTLP exporter: %w", err)
    }

    // 建立 Resource
    res := resource.NewWithAttributes(
        semconv.SchemaURL,
        semconv.ServiceName(cfg.App.Name),
        semconv.ServiceVersion("1.0.0"),
        semconv.DeploymentEnvironment(cfg.App.Env),
    )

    // 設置導出間隔
    interval := cfg.Metrics.OTLP.Interval
    if interval == 0 {
        interval = 60 * time.Second
    }

    // 建立 MeterProvider
    m.meterProvider = sdkmetric.NewMeterProvider(
        sdkmetric.WithResource(res),
        sdkmetric.WithReader(
            sdkmetric.NewPeriodicReader(exporter, sdkmetric.WithInterval(interval)),
        ),
    )

    // 設置全域 MeterProvider
    otel.SetMeterProvider(m.meterProvider)

    // 建立 Meter
    meter := m.meterProvider.Meter("fat-notification-cat")

    // 初始化所有 metrics
    if err = m.initMetrics(meter); err != nil {
        return nil, fmt.Errorf("failed to init metrics: %w", err)
    }

    log.Info("metrics service initialized successfully",
        logger.String("endpoint", cfg.Metrics.OTLP.Endpoint),
        logger.String("path", cfg.Metrics.OTLP.Path),
        logger.Duration("interval", interval))

    return m, nil
}
```

#### 2.4.2 Exporter 建立

```go
// createExporter 建立 OTLP HTTP Exporter
func (m *Metrics) createExporter(cfg *config.Config) (sdkmetric.Exporter, error) {
    // 處理 endpoint，移除協議前綴
    endpoint := cfg.Metrics.OTLP.Endpoint
    isHTTPS := true
    if strings.HasPrefix(endpoint, "https://") {
        endpoint = strings.TrimPrefix(endpoint, "https://")
    } else if strings.HasPrefix(endpoint, "http://") {
        endpoint = strings.TrimPrefix(endpoint, "http://")
        isHTTPS = false
    }

    // 建立自定義 HTTP Client，強制使用 HTTP/1.1 避免 HTTP/2 GOAWAY 問題
    httpClient := &http.Client{
        Transport: &http.Transport{
            TLSClientConfig: &tls.Config{
                MinVersion: tls.VersionTLS12,
            },
            ForceAttemptHTTP2: false,
            MaxIdleConns:        10,
            MaxIdleConnsPerHost: 10,
            IdleConnTimeout:     30 * time.Second,
        },
        Timeout: cfg.Metrics.OTLP.Timeout,
    }

    opts := []otlpmetrichttp.Option{
        otlpmetrichttp.WithEndpoint(endpoint),
        otlpmetrichttp.WithHTTPClient(httpClient),
    }

    // 設置 URL Path
    if cfg.Metrics.OTLP.Path != "" {
        opts = append(opts, otlpmetrichttp.WithURLPath(cfg.Metrics.OTLP.Path))
    }

    // 設置是否使用不安全連線
    if cfg.Metrics.OTLP.Insecure || !isHTTPS {
        opts = append(opts, otlpmetrichttp.WithInsecure())
    }

    // 設置 Headers
    if len(cfg.Metrics.OTLP.Headers) > 0 {
        opts = append(opts, otlpmetrichttp.WithHeaders(cfg.Metrics.OTLP.Headers))
    }

    // 設置 Timeout
    if cfg.Metrics.OTLP.Timeout > 0 {
        opts = append(opts, otlpmetrichttp.WithTimeout(cfg.Metrics.OTLP.Timeout))
    }

    return otlpmetrichttp.New(context.Background(), opts...)
}
```

## 4. 整合點

### 4.1 Web Handler Middleware

建立 metrics middleware 來自動記錄 API 請求：

```go
// internal/adapter/inbound/middleware/metrics.go
func MetricsMiddleware(m *metrics.Metrics) gin.HandlerFunc {
    return func(c *gin.Context) {
        start := time.Now()

        c.Next()

        duration := time.Since(start).Seconds()
        status := strconv.Itoa(c.Writer.Status())

        m.RecordAPIRequest(c.Request.Method, c.FullPath(), status)
        m.RecordAPIDuration(c.Request.Method, c.FullPath(), duration)

        if c.Writer.Status() >= 400 {
            m.RecordAPIError(c.Request.Method, c.FullPath(), status)
        }
    }
}
```

### 4.2 UseCase 整合

在關鍵 UseCase 中注入 Metrics：

```go
type MessageCampaignUseCase struct {
    // ... existing fields
    metrics *metrics.Metrics
}

func (u *MessageCampaignUseCase) ProcessCampaign(ctx context.Context, campaign *entity.MessageCampaign) error {
    start := time.Now()
    merchantID := strconv.FormatUint(campaign.MerchantID, 10)

    err := u.doProcessCampaign(ctx, campaign)

    duration := time.Since(start).Seconds()
    u.metrics.RecordCampaignDuration(merchantID, duration)

    if err != nil {
        u.metrics.RecordCampaignError(merchantID, "process_failed")
        return err
    }

    u.metrics.RecordCampaignProcessed(merchantID, campaign.ObjectType)
    return nil
}
```

### 4.3 Consumer 整合

```go
func (c *Consumer) processEvent(ctx context.Context, event *Event) error {
    start := time.Now()

    c.metrics.RecordEventReceived(event.Type)

    err := c.handleEvent(ctx, event)

    duration := time.Since(start).Seconds()
    c.metrics.RecordEventDuration(event.Type, duration)
    c.metrics.RecordEventProcessed(event.Type, err == nil)

    return err
}
```


### 8.2 高基數標籤警告

避免使用高基數標籤（如 player_id、message_id），這會導致 metrics 膨脹。建議使用：
- merchant_id（商戶數有限）
- campaign_type（類型有限）
- event_type（事件類型有限）
- status codes（HTTP 狀態碼有限）

### 8.3 效能考量

- Metrics 記錄應為非阻塞操作
- 使用 `context.Background()` 避免 context 取消影響 metrics
- 導出間隔預設 60 秒，可根據需求調整
