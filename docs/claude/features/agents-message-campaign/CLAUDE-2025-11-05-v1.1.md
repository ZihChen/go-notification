```go
// SendMessageToCampaignTargets - 代理訊息發送業務流程 (UseCase協調範例)
func (uc *agentUseCase) SendMessageToCampaignTargets(ctx context.Context, campaign *entity.AgentCampaign) error {
    // UseCase職責：協調整個發送流程，整合多個Service和Repository
    
    // 1. 業務規則：解析目標代理類型
    var targetAgents []string
    var err error
    
    switch campaign.TargetType {
    case "all":
        // 簡單查詢：直接使用Repository
        targetAgents, err = uc.getAllActiveAgents(ctx, campaign.MerchantID)
    case "specific":
        // 直接解析：不需要Service
        targetAgents = strings.Split(campaign.TargetDetails, ",")
    case "line":
        // 複雜邏輯：委託Service處理代理線查詢
        targetAgents, err = uc.agentService.GetAgentLineDescendants(campaign.TargetDetails)
    }
    
    if err != nil {
        return fmt.Errorf("failed to resolve target agents: %w", err)
    }
    
    // 2. 業務規則：活躍度檢查 (1個月規則)
    activeAgents := uc.filterActiveAgents(targetAgents)
    
    // 3. Repository協調：批量創建訊息
    if err := uc.createMessagesForAgents(ctx, campaign, activeAgents); err != nil {
        return fmt.Errorf("failed to create messages: %w", err)
    }
    
    // 4. 更新統計數據
    campaign.RealSentCount = int64(len(activeAgents))
    return uc.campaignRepo.Update(ctx, campaign)
}

// filterActiveAgents - 業務規則：活躍度檢查
func (uc *agentUseCase) filterActiveAgents(agentIDs []string) []string {
    // UseCase職責：執行業務規則
    threshold := time.Now().AddDate(0, -1, 0) // 1個月前
    var activeAgents []string
    
    for _, agentID := range agentIDs {
        agent, err := uc.agentRepo.FindByGlobalID(agentID)
        if err != nil {
            continue
        }
        
        // 業務規則：1個月內活躍的代理
        if agent.CurrentSignInAt != nil && agent.CurrentSignInAt.After(threshold) {
            activeAgents = append(activeAgents, agentID)
        }
    }
    
    return activeAgents
}

// syncCurrentAgentUpsert - 同步當前代理資料 (冪等性 Upsert)
func (uc *agentUseCase) syncCurrentAgentUpsert(ctx context.Context, event *AgentSyncEvent) error {
    agent := &entity.Agent{
        MerchantID:      event.Merchant.ID,
        GlobalAgentID:   event.Agent.GlobalAgentID,
        Account:         event.Agent.Account,
        Ancestry:        event.Agent.Ancestry,
        CurrentSignInAt: nil, // 從 KDS 事件中獲取，如果沒有則為 nil
        CreatedAt:       event.Agent.CreatedAt,
        UpdatedAt:       event.Agent.UpdatedAt,
    }
    
    // 使用 global_agent_id 作為 unique key 的冪等性 Upsert
    return uc.agentRepo.Upsert(ctx, agent)
}

// ensureParentAgentsExistUpsert - 確保所有父代理存在 (冪等性 Upsert)
func (uc *agentUseCase) ensureParentAgentsExistUpsert(ctx context.Context, event *AgentSyncEvent) error {
    // 解析 ancestry 路徑
    ancestryParts := strings.Split(event.Agent.Ancestry, "/")
    if len(ancestryParts) <= 1 {
        return nil // 頂級代理，無需檢查父代理
    }
    
    // 為每個父代理執行冪等性 Upsert
    for _, parentGlobalID := range ancestryParts[:len(ancestryParts)-1] {
        if parentGlobalID == "" {
            continue
        }
        
        // 創建基本父代理資料 (如果不存在)
        parentAgent := &entity.Agent{
            MerchantID:    event.Merchant.ID,
            GlobalAgentID: parentGlobalID,
            Account:       extractAccountFromGlobalID(parentGlobalID), // 從 global_id 提取 account
            Ancestry:      "",     // 空值，等待後續同步
            CurrentSignInAt: nil, // 空值，等待實際登入時間同步
            CreatedAt:     time.Now(),
            UpdatedAt:     time.Now(), // 重要：使用當前時間作為基準
        }
        
        // 使用冪等性 Upsert，global_agent_id 作為 unique key
        if err := uc.agentRepo.Upsert(ctx, parentAgent); err != nil {
            return fmt.Errorf("failed to upsert parent agent %s: %w", parentGlobalID, err)
        }
        
        uc.logger.Info("Upserted parent agent",
            logger.Field{Key: "parent_global_id", Value: parentGlobalID},
            logger.Field{Key: "merchant_id", Value: event.Merchant.ID})
    }
    
    return nil
}

// extractAccountFromGlobalID - 從 global_id 提取 account
func extractAccountFromGlobalID(globalID string) string {
    // 範例: "PHW-AGENT-123" -> "agent123" 或其他提取邏輯
    parts := strings.Split(globalID, "-")
    if len(parts) >= 3 {
        return strings.ToLower(parts[1]) + parts[2] // "agent123"
    }
    return globalID // fallback
}
```

##### 2.3.2 AgentService 實作 (領域邏輯服務)
```go
// internal/application/service/agent_service.go

type AgentService struct {
    agentRepo ports.AgentRepository

// SyncAgentRelationshipsUpsert - 複雜的代理關聯同步邏輯 (Service職責)
func (s *AgentService) SyncAgentRelationshipsUpsert(ctx context.Context, event *AgentSyncEvent) error {
    // Service職責：處理複雜的領域邏輯
    agent := event.Agent
    
    // 複雜的ancestry解析邏輯 (領域知識)
    pathParts := strings.Split(agent.Ancestry, "/")
    if len(pathParts) < 2 {
        return nil // 頂級代理，無需建立關係
    }
    
    // 建立每一層的父子關係 (領域專業邏輯)
    for i := 0; i < len(pathParts)-1; i++ {
        parentGlobalID := pathParts[i]
        childGlobalID := pathParts[i+1]
        
        // ID轉換邏輯 (領域轉換)
        parentID, err := s.agentRepo.GetAgentIDByGlobalID(parentGlobalID, agent.MerchantID)
        if err != nil {
            s.logger.Warn("Parent agent not found", 
                logger.Field{Key: "parent_global_id", Value: parentGlobalID})
            continue
        }
        
        childID, err := s.agentRepo.GetAgentIDByGlobalID(childGlobalID, agent.MerchantID)
        if err != nil {
            s.logger.Warn("Child agent not found", 
                logger.Field{Key: "child_global_id", Value: childGlobalID})
            continue
        }
        
        // 領域物件創建與複雜計算
        relationship := &entity.AgentRelationship{
            ParentID:   parentID,
            ChildID:    childID,
            DepthLevel: i + 1,
            PathHash:   s.generatePathHash(pathParts[:i+2]), // 複雜計算邏輯
            MerchantID: agent.MerchantID,
        }
        
        if err := s.agentRepo.UpsertRelationship(ctx, relationship); err != nil {
            return fmt.Errorf("failed to upsert relationship: %w", err)
        }
    }
    
    return nil
}

// GetAgentLineDescendants - 複雜的代理線查詢邏輯 (Service職責)
func (s *AgentService) GetAgentLineDescendants(globalAgentID string) ([]string, error) {
    // Service職責：處理複雜查詢邏輯與效能優化
    
    // 1. 快取策略 (效能優化邏輯)
    if cached := s.getFromCache(globalAgentID); cached != nil {
        s.logger.Debug("Agent line cache hit", 
            logger.Field{Key: "agent_id", Value: globalAgentID})
        return cached, nil
    }
    
    // 2. 複雜的代理樹查詢邏輯 (領域知識)
    agents, err := s.agentRepo.QueryAgentsByPath(globalAgentID)
    if err != nil {
        return nil, fmt.Errorf("failed to query agent descendants: %w", err)
    }
    
    // 3. 結果驗證與處理 (領域規則)
    validAgents := s.validateAndFilterAgents(agents)
    
    // 4. 快取管理策略 (效能優化)
    s.setCache(globalAgentID, validAgents)
    
    s.logger.Info("Agent line query completed",
        logger.Field{Key: "agent_id", Value: globalAgentID},
        logger.Field{Key: "result_count", Value: len(validAgents)})
    
    return validAgents, nil
}

    // 並發查詢後代 (效能優化邏輯)
    go func() {
        descendants, err2 = s.GetAgentLineDescendants(globalAgentID)
        done <- struct{}{}
    }()
    
    // 等待兩個查詢完成
    <-done
    <-done
    
    if err1 != nil {
        return nil, fmt.Errorf("failed to get ancestors: %w", err1)
    }
    if err2 != nil {
        return nil, fmt.Errorf("failed to get descendants: %w", err2)
    }
    
    // 領域物件組裝
    return &AgentHierarchy{
        AgentID:     globalAgentID,
        Ancestors:   ancestors,
        Descendants: descendants,
        TotalCount:  len(ancestors) + len(descendants) + 1,
    }, nil
}

// generatePathHash - 複雜的領域計算邏輯 (Service職責)
func (s *AgentService) generatePathHash(pathParts []string) string {
    // Service職責：演算法實作與領域計算
    pathStr := strings.Join(pathParts, "/")
    hash := sha256.Sum256([]byte(pathStr))
    return hex.EncodeToString(hash[:])[:16] // 取前16字符
}

// validateAndFilterAgents - 領域驗證邏輯 (Service職責)
func (s *AgentService) validateAndFilterAgents(agents []string) []string {
    // Service職責：領域規則驗證
    var validAgents []string
    
    for _, agentID := range agents {
        // 領域驗證：ID格式檢查
        if s.isValidAgentID(agentID) {
            validAgents = append(validAgents, agentID)
        }
    }
    
    return validAgents
}

// isValidAgentID - 領域知識：代理ID格式驗證
func (s *AgentService) isValidAgentID(agentID string) bool {
    // Service職責：領域知識封裝
    return strings.HasPrefix(agentID, "FATCAT-AGENT-") || 
           strings.HasPrefix(agentID, "PHW-AGENT-")
}
```
