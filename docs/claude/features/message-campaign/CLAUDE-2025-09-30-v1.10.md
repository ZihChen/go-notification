```go
// 新的高效能實作
func (r *CampaignTargetRepository) FindCampaignIDsByPlayerCriteria(
    ctx context.Context,
    playerAccount string,
    levelID string,
    tagIDs []uint64,
    merchantID uint64,
) ([]uint64, error) {
    
    tagIDStrings := make([]string, len(tagIDs))
    for i, id := range tagIDs {
        tagIDStrings[i] = fmt.Sprintf("%d", id)
    }
    
    query := `
        SELECT DISTINCT ct.campaign_id 
        FROM campaign_targets ct
        INNER JOIN message_campaigns mc ON ct.campaign_id = mc.id
        WHERE mc.merchant_id = ? 
          AND mc.status IN ('scheduled', 'sent')
          AND mc.deleted_at IS NULL
          AND mc.created_at <= NOW()
          AND (
            (ct.target_type = 'all') OR
            (ct.target_type = 'player' AND ct.target_id = ?) OR
            (ct.target_type = 'level' AND ct.target_id = ?) OR
            (ct.target_type = 'tag' AND ct.target_id IN (?))
          )
    `
    
    var campaignIDs []uint64
    err := r.db.WithContext(ctx).Raw(query, 
        merchantID, playerAccount, levelID, tagIDStrings,
    ).Scan(&campaignIDs).Error
    
    return campaignIDs, err
}
```

```go
// internal/application/usecase/message/message_usecase.go
func (u *MessageUseCase) ProcessPlayerV2(ctx context.Context, globalPlayerID string) error {
    ctx, span := u.tracingService.StartSpan(ctx, "MessageUseCase.ProcessPlayerV2")
    defer u.tracingService.SpanEnd(span)

    // 獲取玩家資訊
    player, err := u.playerRepo.FindByGlobalID(ctx, globalPlayerID)
    if err != nil {
        u.tracingService.RecordSpanError(span, err)
        return fmt.Errorf("find player: %w", err)
    }

    // 獲取玩家標籤IDs
    playerTagIDs, err := u.playerRepo.GetPlayerTagIDs(ctx, player.ID)
    if err != nil {
        u.tracingService.RecordSpanError(span, err)
        return fmt.Errorf("get player tag IDs: %w", err)
    }

    // 一次性查詢所有符合條件的 campaigns
    eligibleCampaignIDs, err := u.campaignTargetRepo.FindCampaignIDsByPlayerCriteria(ctx, 
        player.Account, 
        fmt.Sprintf("%d", player.LevelID), 
        playerTagIDs,
        player.MerchantID,
    )
    if err != nil {
        u.tracingService.RecordSpanError(span, err)
        return fmt.Errorf("find eligible campaigns: %w", err)
    }

    u.tracingService.RecordSpanAttributes(span, 
        attribute.Int("eligible_campaigns", len(eligibleCampaignIDs)))

    // 批量處理 player_message 創建
    return u.processCampaignMessages(ctx, globalPlayerID, eligibleCampaignIDs)
}

func (u *MessageUseCase) processCampaignMessages(
    ctx context.Context, 
    globalPlayerID string, 
    campaignIDs []uint64,
) error {
    if len(campaignIDs) == 0 {
        return nil
    }

    // 檢查已存在的訊息
    existingMessageCampaignIDs, err := u.playerMessageRepo.FindExistingCampaignIDs(
        ctx, globalPlayerID, campaignIDs,
    )
    if err != nil {
        return fmt.Errorf("find existing messages: %w", err)
    }

    // 計算需要創建的新訊息
    newCampaignIDs := make([]uint64, 0, len(campaignIDs))
    existingSet := make(map[uint64]bool)
    for _, id := range existingMessageCampaignIDs {
        existingSet[id] = true
    }

    for _, id := range campaignIDs {
        if !existingSet[id] {
            newCampaignIDs = append(newCampaignIDs, id)
        }
    }

    if len(newCampaignIDs) == 0 {
        return nil
    }

    // 批量創建新訊息
    newMessages := make([]*entity.PlayerMessage, len(newCampaignIDs))
    for i, campaignID := range newCampaignIDs {
        newMessages[i] = &entity.PlayerMessage{
            GlobalPlayerID: globalPlayerID,
            CampaignID:     campaignID,
            IsRead:         false,
            CreatedAt:      time.Now(),
        }
    }

    if err = u.playerMessageRepo.CreateBatch(ctx, newMessages); err != nil {
        return fmt.Errorf("create batch messages: %w", err)
    }

    u.logger.InfoLog("New messages created for returning player",
        u.logger.String("global_player_id", globalPlayerID),
        u.logger.Int("new_messages_count", len(newMessages)))

    return nil
}
```

### Phase 3: 資料遷移策略

#### 3.1 遷移腳本
```go
// cmd/migrate/campaign_targets.go
func MigrateCampaignTargets(ctx context.Context, db *gorm.DB) error {
    var campaigns []models.MessageCampaign
    if err := db.WithContext(ctx).Find(&campaigns).Error; err != nil {
        return err
    }

    for _, campaign := range campaigns {
        targets, err := parseCampaignTargets(campaign.Target, campaign.TargetDetail)
        if err != nil {
            log.Printf("Failed to parse targets for campaign %d: %v", campaign.ID, err)
            continue
        }

        for _, target := range targets {
            target.CampaignID = campaign.ID
            if err := db.WithContext(ctx).Create(target).Error; err != nil {
                log.Printf("Failed to create target for campaign %d: %v", campaign.ID, err)
            }
        }
    }

    return nil
}

func parseCampaignTargets(targetType string, targetDetail *string) ([]*models.CampaignTarget, error) {
    var targets []*models.CampaignTarget

    switch targetType {
    case consts.TargetAll:
        targets = append(targets, &models.CampaignTarget{
            TargetType: consts.TargetAll,
            TargetID:   nil,
        })
    case consts.TargetPlayer:
        if targetDetail == nil {
            return nil, fmt.Errorf("target detail is required for player target")
        }
        var accounts []string
        if err := json.Unmarshal([]byte(*targetDetail), &accounts); err != nil {
            return nil, err
        }
        for _, account := range accounts {
            targets = append(targets, &models.CampaignTarget{
                TargetType: consts.TargetPlayer,
                TargetID:   &account,
            })
        }
    case consts.TargetLevel:
        if targetDetail == nil {
            return nil, fmt.Errorf("target detail is required for level target")
        }
        var levelIDs []string
        if err := json.Unmarshal([]byte(*targetDetail), &levelIDs); err != nil {
            return nil, err
        }
        for _, levelID := range levelIDs {
            targets = append(targets, &models.CampaignTarget{
                TargetType: consts.TargetLevel,
                TargetID:   &levelID,
            })
        }
    case consts.TargetTag:
        if targetDetail == nil {
            return nil, fmt.Errorf("target detail is required for tag target")
        }
        var tagIDs []string
        if err := json.Unmarshal([]byte(*targetDetail), &tagIDs); err != nil {
            return nil, err
        }
        for _, tagID := range tagIDs {
            targets = append(targets, &models.CampaignTarget{
                TargetType: consts.TargetTag,
                TargetID:   &tagID,
            })
        }
    }

    return targets, nil
}
```

```go
// internal/application/usecase/message/message_usecase.go
func (u *MessageUseCase) CreateMessageCampaign(
    ctx context.Context, 
    req *dto.CreateMessageCampaignRequest,
) error {
    // 創建主記錄
    campaign := &entity.MessageCampaign{...}
    if err := u.campaignRepo.Create(ctx, campaign); err != nil {
        return err
    }

    // 同步創建關聯記錄
    targets, err := u.buildCampaignTargets(campaign.Target, req.TargetDetail, campaign.ID)
    if err != nil {
        return fmt.Errorf("build campaign targets: %w", err)
    }

    if err := u.campaignTargetRepo.CreateBatch(ctx, targets); err != nil {
        // 如果關聯記錄創建失敗，回滾主記錄
        u.campaignRepo.Delete(ctx, campaign.ID)
        return fmt.Errorf("create campaign targets: %w", err)
    }

    return nil
}
```
